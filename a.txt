
#!/usr/bin/env bash
set -Eeuo pipefail

HOST="127.0.0.1"
PORT="1443"

DIR="${XDG_STATE_HOME:-$HOME/.local/state}/tg-ws-proxy-launcher"
SECRET_FILE="$DIR/secret"
PID_FILE="$DIR/proxy.pid"
LOG_FILE="$DIR/proxy.log"

mkdir -p "$DIR"

notify() {
    command -v notify-send >/dev/null && notify-send "TG WS Proxy" "$1" || true
}

# Проверяем зависимости
for cmd in tg-ws-proxy ss xdg-open openssl; do
    if ! command -v "$cmd" >/dev/null; then
        notify "Не найдена команда: $cmd"
        exit 1
    fi
done

# Постоянный секрет
if [[ ! -s "$SECRET_FILE" ]]; then
    openssl rand -hex 16 > "$SECRET_FILE"
    chmod 600 "$SECRET_FILE"
fi

SECRET="$(cat "$SECRET_FILE")"

# Проверяем, запущен ли наш прокси
RUNNING=0

if [[ -s "$PID_FILE" ]]; then
    PID="$(cat "$PID_FILE")"

    if kill -0 "$PID" 2>/dev/null &&
       ps -p "$PID" -o args= | grep -Fq "tg-ws-proxy"; then
        RUNNING=1
    else
        rm -f "$PID_FILE"
    fi
fi

# Запускаем прокси
if [[ "$RUNNING" -eq 0 ]]; then
    if ss -H -ltn "sport = :$PORT" | grep -q .; then
        notify "Порт $PORT уже занят!"
        exit 1
    fi

    nohup tg-ws-proxy \
        --host "$HOST" \
        --port "$PORT" \
        --secret "$SECRET" \
        --log-file "$LOG_FILE" \
        >> "$LOG_FILE" 2>&1 </dev/null &

    echo $! > "$PID_FILE"
fi

# Ждём готовности сервера
READY=0

for _ in {1..40}; do
    if ss -H -ltn "sport = :$PORT" | grep -q .; then
        READY=1
        break
    fi

    sleep 0.25
done

if [[ "$READY" -ne 1 ]]; then
    notify "Прокси не запустился. Проверь $LOG_FILE"
    exit 1
fi

# Передаём Telegram ссылку на локальный MTProto-прокси
URL="tg://proxy?server=$HOST&port=$PORT&secret=dd$SECRET"

xdg-open "$URL"

notify "Прокси запущен. Подтверди добавление в Telegram."
