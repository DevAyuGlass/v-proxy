
import pygame
import random
import sys

pygame.init()

CELL = 25
WIDTH, HEIGHT = 800, 600
COLS, ROWS = WIDTH // CELL, HEIGHT // CELL

screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption("Snake 🐍")

clock = pygame.time.Clock()
font = pygame.font.SysFont("sans", 32)

BLACK = (18, 18, 24)
GREEN = (80, 220, 120)
RED = (240, 70, 70)
WHITE = (240, 240, 240)

def random_food(snake):
    while True:
        food = (random.randrange(COLS), random.randrange(ROWS))
        if food not in snake:
            return food

snake = [(10, 10), (9, 10), (8, 10)]
direction = (1, 0)
next_direction = direction
food = random_food(snake)
score = 0
game_over = False

while True:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            pygame.quit()
            sys.exit()

        if event.type == pygame.KEYDOWN:
            keys = {
                pygame.K_UP: (0, -1),
                pygame.K_DOWN: (0, 1),
                pygame.K_LEFT: (-1, 0),
                pygame.K_RIGHT: (1, 0),
                pygame.K_w: (0, -1),
                pygame.K_s: (0, 1),
                pygame.K_a: (-1, 0),
                pygame.K_d: (1, 0),
            }

            if event.key == pygame.K_r and game_over:
                snake = [(10, 10), (9, 10), (8, 10)]
                direction = next_direction = (1, 0)
                food = random_food(snake)
                score = 0
                game_over = False

            if event.key in keys and not game_over:
                new_dir = keys[event.key]
                if new_dir != (-direction[0], -direction[1]):
                    next_direction = new_dir

    if not game_over:
        direction = next_direction
        head = (
            snake[0][0] + direction[0],
            snake[0][1] + direction[1]
        )

        eating = head == food

        if (
            head[0] < 0 or head[0] >= COLS
            or head[1] < 0 or head[1] >= ROWS
            or head in (snake if eating else snake[:-1])
        ):
            game_over = True
        else:
            snake.insert(0, head)

            if eating:
                score += 1
                food = random_food(snake)
            else:
                snake.pop()

    screen.fill(BLACK)

    for x, y in snake:
        pygame.draw.rect(
            screen, GREEN,
            (x * CELL, y * CELL, CELL - 2, CELL - 2),
            border_radius=5
        )

    pygame.draw.rect(
        screen, RED,
        (food[0] * CELL, food[1] * CELL, CELL - 2, CELL - 2),
        border_radius=8
    )

    text = font.render(f"Score: {score}", True, WHITE)
    screen.blit(text, (15, 10))

    if game_over:
        msg = font.render("GAME OVER - Press R", True, WHITE)
        rect = msg.get_rect(center=(WIDTH // 2, HEIGHT // 2))
        screen.blit(msg, rect)

    pygame.display.flip()
    clock.tick(10 + min(score // 5, 15))
